63def complete_ean13(value): 

64 """Добавить или проверить контрольную цифру GTIN-13.""" 

65 if not isinstance(value, str) or not value.isascii(): 

66 raise ValueError("ean13 должен быть строкой из 12 или 13 цифр") 

67 if not value.isdigit() or len(value) not in (DATA_LENGTH, CODE_LENGTH): 

68 raise ValueError("ean13 должен содержать 12 или 13 цифр") 

69 data = value[:DATA_LENGTH] 

70 total = sum( 

71 int(digit) * (1 if index % 2 == EVEN_REMAINDER else 3) 

72 for index, digit in enumerate(data) 

73 ) 

74 check = str(-total % 10) 

75 if len(value) == CODE_LENGTH and value[-1] != check: 

76 raise ValueError("Неверная контрольная цифра EAN-13") 

77 return data + check

80def encode_modules(value): 

81 """Преобразовать 13 цифр в 95 светлых и тёмных модулей.""" 

82 digits = complete_ean13(value) 

83 first = int(digits[0]) 

84 left = "".join( 

85 (LEFT if parity == "L" else MIXED)[int(digit)] 

86 for parity, digit in zip(PARITY[first], digits[1:7]) 

87 ) 

88 right = "".join(RIGHT[int(digit)] for digit in digits[7:]) 

89 return "101" + left + "01010" + right + "101"

92def render_svg(value): 

93 """Построить масштабируемый SVG с тихими зонами и подписью.""" 

94 digits = complete_ean13(value) 

95 bits = encode_modules(digits) 

96 bars = [] 

97 position = LEFT_QUIET 

98 for bit, group in groupby(bits): 

99 width = len(list(group)) 

100 if bit == "1": 

101 bars.append( 

102 f'<rect x="{position}" y="0" width="{width}" ' 

103 f'height="{BAR_HEIGHT}"/>' 

104 ) 

105 position += width 

106 content = "".join(bars) 

107 return ( 

108 '<svg xmlns="http://www.w3.org/2000/svg" ' 

109 f'viewBox="0 0 {SVG_WIDTH} {SVG_HEIGHT}" ' 

110 'shape-rendering="crispEdges">' 

111 f"<title>EAN-13 {digits}</title>{content}" 

112 f'<text x="{LEFT_QUIET + SYMBOL_WIDTH // 2}" y="68" ' 

113 'text-anchor="middle" font-family="sans-serif" ' 

114 f'font-size="9">{digits}</text></svg>' 

115 )

118def generate(payload): 

119 """Принять JSON-объект с числом EAN-13 и вернуть SVG.""" 

120 if not isinstance(payload, dict) or set(payload) != {"ean13"}: 

121 raise ValueError("Ожидается объект JSON с полем ean13") 

122 return render_svg(payload["ean13"])
